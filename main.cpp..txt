#include <QApplication>
#include <QMainWindow>
#include <QLabel>

class MainWindow : public QMainWindow {
public:
    MainWindow() {
        QLabel *label = new QLabel("TakeoffWorks Native - Phase 1", this);
        label->setAlignment(Qt::AlignCenter);
        setCentralWidget(label);
        resize(400, 200);
        setWindowTitle("TakeoffWorks Native");
    }
};

int main(int argc, char *argv[]) {
    QApplication app(argc, argv);
    MainWindow w;
    w.show();
    return app.exec();
}
