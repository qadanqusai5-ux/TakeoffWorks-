cmake_minimum_required(VERSION 3.16)
project(TakeoffWorks_Native)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)

find_package(Qt6 REQUIRED COMPONENTS Core Widgets Gui)

# PDFium Setup (دليل حيث سيتم تنزيله)
set(PDFium_DIR "C:/deps/pdfium" CACHE PATH "Path to PDFium")
if(EXISTS ${PDFium_DIR})
    find_library(PDFium_LIBRARY NAMES pdfium PATHS ${PDFium_DIR})
    find_path(PDFium_INCLUDE_DIR NAMES fpdfview.h PATHS ${PDFium_DIR}/include)
    if(PDFium_LIBRARY AND PDFium_INCLUDE_DIR)
        message(STATUS "PDFium found.")
    else()
        message(WARNING "PDFium not found. The app will still compile but PDF functions won't work.")
    endif()
endif()

add_executable(TakeoffWorks_Native main.cpp)
target_link_libraries(TakeoffWorks_Native Qt6::Core Qt6::Gui Qt6::Widgets)

if(PDFium_LIBRARY)
    target_link_libraries(TakeoffWorks_Native ${PDFium_LIBRARY})
    target_include_directories(TakeoffWorks_Native PRIVATE ${PDFium_INCLUDE_DIR})
endif()
